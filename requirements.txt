# Flow Studio — Python dependencies (Windows, Python 3.12)
# Install:  py -3.12 -m venv venv && venv\Scripts\pip install -r requirements.txt
#
# Only direct dependencies are pinned; pip resolves the rest (torch's tree,
# misaki, ctranslate2, av, …). Versions match the tested working environment.

# --- Text to Speech (app.py) ---
kokoro==0.9.4          # Kokoro-82M TTS; pulls torch + misaki (G2P). Downloads ~330 MB model on first use.
soundfile==0.14.0      # WAV read/write
# misaki's English G2P loads the spaCy model en_core_web_sm. It is NOT a normal PyPI
# package, so it must be pinned as a direct wheel — otherwise misaki falls back to
# `spacy.cli.download`, which shells to pip, which a uv-created venv does not have.
en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.8.0/en_core_web_sm-3.8.0-py3-none-any.whl

# --- Dictation (flow.py) ---
faster-whisper==1.2.1  # local ASR; pulls ctranslate2. Downloads the Whisper model on first use (~244 MB for small).
sounddevice==0.5.5     # microphone capture
keyboard==0.13.5; sys_platform == "win32"  # ctrl+v paste into the focused app (hotkey itself uses Win32 RegisterHotKey)
# macOS hotkey/paste (os_mac.py) use pyobjc AppKit + Quartz, which pywebview installs on the Mac.

# --- Desktop shell (flow_studio.py) ---
pywebview==6.2.1       # native WebView2 window; falls back to the browser if unavailable
pystray==0.19.5        # system-tray icon (close/quit from the bottom-right menu)
pillow==12.3.0         # tray icon image (pystray loads flow.ico via PIL)
# Linux window: pywebview's Qt backend. The frozen app carries Qt WebEngine itself (bundling
# WebKitGTK breaks across distros); PySide6 over PyQt6 because it's LGPL.
PySide6==6.9.3; sys_platform == "linux"
qtpy==2.4.3; sys_platform == "linux"

# --- Shared ---
Flask==3.1.3           # local web servers (TTS :7500, dictation :7600)
numpy==2.5.1

# torch is a transitive dep of kokoro, but pin it explicitly: it dominates the
# install size (~1-2 GB) and is the main thing that breaks a PyInstaller build,
# so a deterministic version matters for reproducible deploys. CPU wheel is the
# PyPI default on Windows.
torch==2.13.0

# kokoro pulls transformers unpinned; without a floor the resolver backtracks to 2.3.0,
# whose tokenizers 0.13.3 has no Python 3.12 wheel (needs a Rust source build).
transformers>=4.48
# faster-whisper 1.2.1 calls av.open(..., metadata_errors=...), which av 19 removed.
av<19

# Text extraction for the TTS tool's "Open PDF" button. Chosen over pypdf for
# layout-aware extraction: it keeps columns and spacing readable, which matters
# because garbled text becomes garbled speech.
pymupdf==1.28.2

# Article extraction for the TTS tool's "Open URL" box (reader mode). Chosen over
# readability-lxml for cleaner output: it drops nav, ads, cookie banners and comments,
# and gives the headline. Optional — the app runs without it, and it only touches the
# network when you paste a URL.
trafilatura==2.2.0

# Not a Python package — required at runtime, install separately (see README):
#   • Ollama + a pulled model (default: qwen2.5:3b)  → LLM cleanup of transcripts
#   • Microsoft Edge WebView2 Runtime               → native desktop window (preinstalled on Win11)
#   • ffmpeg on PATH (optional)                      → MP3 export in the TTS tool
